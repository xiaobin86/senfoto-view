# Senfoto008 实时 RPM 显示 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 让 Senfoto008 机械雷达在 SenFoToView 中显示实时转速（RPM），复用上游 `vtkRPMText` 过滤器，仅在 Senfoto008 源上生效。

**Architecture:** 在 `vtkSenfoto008PacketInterpreter` 内**内联**计算 `Rpm`/`Frequency`（方位角/时间极值，复用上游 `vtkRPMText` 过滤器显示；**不改**第三方子模块 `VelodynePlugin`）；基类 `SplitFrame()` 自动写 field data `"RPM"/"FPS"`；打开 Senfoto008 源时自动挂载 `RPMText`，并提供 Tools 菜单开关。

> **执行时修订（2026-10-17）**：`Plugins/VelodynePlugin` 是第三方 git submodule，**不可迁移/复用**其 `RPMCalculator`。**原 Task 1（迁移 RPMCalculator）作废**；RPM/FPS 改为在 `vtkSenfoto008PacketInterpreter` **内联实现**（见 spec §4.1/§4.2）。下文 Task 2 的执行改为内联（不含 `RPMCalculator` 成员/头文件依赖），Task 3/4/5 不变。

**Tech Stack:** C++ / VTK、ParaView ServerManager XML、CMake (`vtk_module_add_module` / `vtk_add_test_cxx`)、Qt (`pqObjectBuilder`, `QSettings`, `pqReaction`)。

**Spec:** `docs/superpowers/specs/2026-10-17-senfoto008-rpm-design.md`

```cpp
// Packet timestamp: offset 20, 6 bytes seconds LE + 4 bytes microseconds LE
// (matches senfoto008::GetPacketTimestamp).
void SetPacketTimestamp(std::vector<unsigned char>& packet, std::uint64_t seconds,
  std::uint32_t microseconds)
{
  unsigned char* ts = packet.data() + 20;
  for (int i = 0; i < 6; ++i)
  {
    ts[i] = static_cast<unsigned char>((seconds >> (8 * i)) & 0xFF);
  }
  ts[6] = static_cast<unsigned char>(microseconds & 0xFF);
  ts[7] = static_cast<unsigned char>((microseconds >> 8) & 0xFF);
  ts[8] = static_cast<unsigned char>((microseconds >> 16) & 0xFF);
  ts[9] = static_cast<unsigned char>((microseconds >> 24) & 0xFF);
}
```

```cpp
  // RPM / Frequency: 3 revolutions, 10 packets each (az 0..359.9 deg),
  // 10 ms per packet  =>  ~1 revolution / 100 ms  =>  RPM ~600, FPS ~10.
  // ---------------------------------------------------------------------------

    rpmInterp->Initialize();

    std::uint64_t sec = 1000;
    std::uint32_t micros = 0;
    for (int rev = 0; rev < 3; ++rev)
    {
      for (int i = 0; i < 10; ++i)
      {

          static_cast<std::uint16_t>(std::lround(i * 35990.0 / 9.0)); // 0 .. 359.90 deg
        auto p = Build48LinePacket(azUnits);
        SetPacketTimestamp(p, sec, micros);
        rpmInterp->ProcessPacket(p.data(), static_cast<unsigned int>(p.size()));
        micros += 10000; // +10 ms
        if (micros >= 1000000) { micros -= 1000000; ++sec; }
      }

      return 1;
    }
    if (!(fps > 8.0 && fps < 12.0))
    {
      std::cerr << "FPS out of range: " << fps << std::endl;
      return 1;
    }
  }
```

把现有这一整段

```cpp
  double blockAzimuths[senfoto008::DATA_BLOCKS] = {};
  int wrapIdx = -1;
  for (std::size_t b = 0; b < senfoto008::DATA_BLOCKS; ++b)
  {
    blockAzimuths[b] = senfoto008::GetBlockAzimuth(data, b);

    {
      wrapIdx = static_cast<int>(b);

```

替换为：

```cpp
  // 包时间戳先取（uint32 微秒，供 RPM 计算）。
  const double packetTimestamp = senfoto008::GetPacketTimestamp(data);
  const unsigned int tsUs = static_cast<unsigned int>(packetTimestamp * 1e6);
  if (!internals->HasT) { internals->TMin = internals->TMax = tsUs; internals->HasT = true; }
  else { internals->TMin = std::min(internals->TMin, tsUs);
         internals->TMax = std::max(internals->TMax, tsUs); }

  double blockAzimuths[senfoto008::DATA_BLOCKS] = {};
  int wrapIdx = -1;
  for (std::size_t b = 0; b < senfoto008::DATA_BLOCKS; ++b)
  {
    blockAzimuths[b] = senfoto008::GetBlockAzimuth(data, b);
    const int azHundredth = static_cast<int>(std::lround(blockAzimuths[b] * 100.0));
    if (!internals->HasAz) { internals->AzMin = internals->AzMax = azHundredth; internals->HasAz = true; }
    else { internals->AzMin = std::min(internals->AzMin, azHundredth);
           internals->AzMax = std::max(internals->AzMax, azHundredth); }
    if (wrapIdx < 0 && this->CheckBlockWrap(blockAzimuths[b]))
    {
      wrapIdx = static_cast<int>(b);
    }
  }
  if (wrapIdx >= 0 && this->CurrentFrame && this->CurrentFrame->GetNumberOfPoints() > 0)
  {
    const double dAngle   = (internals->AzMax - internals->AzMin) / 36000.0; // 圈数
    const double dTimeMin = (internals->TMax - internals->TMin) / 60e6;       // 分钟
    this->Rpm = (dTimeMin > 1e-12 && dAngle > 0.0) ? (dAngle / dTimeMin) : 0.0;
    if (internals->HasLastFrameTs && tsUs != internals->LastFrameTsUs)
    {
      this->Frequency = 1.0 / ((tsUs - internals->LastFrameTsUs) * 1e-6);
    }
    internals->LastFrameTsUs = tsUs;
    internals->HasLastFrameTs = true;
    internals->HasAz = false;   // 复位本帧累计
    internals->HasT = false;
    Superclass::SplitFrame();
  }
```

```cpp
// ============================================================
// 功能：Tools 菜单「显示 RPM」开关实现。
// 作者：acelan
// 新建时间：2026-10-17
// 修改时间：2026-10-17
// ============================================================

#include "lqRPMTextReaction.h"

#include <pqActiveObjects.h>
#include <pqApplicationCore.h>
#include <pqObjectBuilder.h>
#include <pqPipelineSource.h>

#include <QAction>
#include <QSettings>

#include <vtkSMProxy.h>

namespace
{
pqPipelineSource* FindRPMTextConsumer(pqPipelineSource* source)
{
  if (!source)
  {
    return nullptr;
  }
  for (pqPipelineSource* consumer : source->getAllConsumers())
  {
    if (consumer && consumer->getProxy() &&
      std::strcmp(consumer->getProxy()->GetXMLName(), "RPMText") == 0)
    {
      return consumer;
    }
  }
  return nullptr;
}

bool IsActiveSenfoto008()
{
  pqPipelineSource* src = pqActiveObjects::instance().activeSource();
  return src && src->getProxy() &&
    QString(src->getProxy()->GetXMLName()).startsWith("Senfoto008");
}
}

//-----------------------------------------------------------------------------
lqRPMTextReaction::lqRPMTextReaction(QAction* parent)
  : Superclass(parent)
{
  QObject::connect(&pqActiveObjects::instance(),
    &pqActiveObjects::sourceChanged,
    this,
    &lqRPMTextReaction::updateEnableState);
  this->updateEnableState();
}

//-----------------------------------------------------------------------------
void lqRPMTextReaction::updateEnableState()
{
  this->parentAction()->setEnabled(::IsActiveSenfoto008());
}

//-----------------------------------------------------------------------------
void lqRPMTextReaction::onTriggered()
{
  pqPipelineSource* source = pqActiveObjects::instance().activeSource();
  if (!source || !source->getProxy() ||
    !QString(source->getProxy()->GetXMLName()).startsWith("Senfoto008"))
  {
    return;
  }

  pqPipelineSource* rpm = ::FindRPMTextConsumer(source);
  if (!rpm)
  {
    pqObjectBuilder* builder = pqApplicationCore::instance()->getObjectBuilder();
    rpm = builder->createFilter("filters", "RPMText", source);
    if (!rpm)
    {
      return;
    }
    rpm->getProxy()->UpdateVTKObjects();
  }

  const bool visible = this->parentAction()->isChecked();
  if (pqView* view = pqActiveObjects::instance().activeView())
  {
    if (pqDataRepresentation* repr = rpm->getRepresentation(view))
    {
      repr->setVisible(visible);
    }
  }
  QSettings().setValue("LidarView/ShowRPM", visible);
  pqApplicationCore::instance()->render();
}
```

（若 `pqView`/`pqDataRepresentation` 未声明，补 `#include <pqView.h>`、`#include <pqDataRepresentation.h>`。）

## 注意 / 风险

- **时间戳可用性**：若实测 `GetPacketTimestamp()` 在一圈内不递增（数据手册 §10.5 曾记录帧 ts 步长为 0），则 Task 2 的 RPM 会异常；此时需回退用「帧计数/系统时间」估算，并把该发现补回 spec。
- **迁移副作用**：Task 1 改了上游 Velodyne `CMakeLists.txt`。若 `HEADERS` 导出非 vtk 类在 IOLidar 中失败，降级为在 `Senfoto008Plugin/.../` 内复制 `RpmCalculator`（并在 vtk.module 加 `#include`），放弃共享。
- **重复挂载**：Task 4 的自动挂载与 Tools 开关都用 find-or-create（`FindRPMTextConsumer`），避免同源出现两个 `RPMText`。
